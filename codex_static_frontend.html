接入示例

复制后在服务端注入 trace_id、span_id 与业务标签，即可进入观测台。

const response = await client.responses.create({
  model: "gpt-4.1",
  input: "分析 AOTU-GPT 最近 15 分钟的工具超时原因",
  metadata: {
    service: "aotu-agent-api",
    env: "prod",
    trace_id: "trc_" + crypto.randomUUID(),
    observer: "aotu-gpt-workbench"
  },
  tools: [
    { type: "function", name: "log.search" },
    { type: "function", name: "metric.query" }
  ],
  stream: true
});
